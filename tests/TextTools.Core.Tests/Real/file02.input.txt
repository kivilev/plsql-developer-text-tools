CREATE OR REPLACE PACKAGE BODY put_util IS

  FUNCTION getSufosuf RETURN VARCHAR2 is

  BEGIN
    RETURN put_const.g_sufosuf;
  END getSufosuf;
--
  PROCEDURE ConnectTo(p_branch VARCHAR2) IS
    v_zakomo VARCHAR2(30);
  BEGIN
    SELECT user_name INTO v_zakomo FROM guhacuz.fe_kohitu_branch WHERE branch = p_branch;
    EXECUTE IMMEDIATE 'cogoc nuzitez nap fahamuf_vehevo = ' || v_zakomo;
    EXECUTE IMMEDIATE ('duhib munufav feg kov_giva_olikid=''ba.un.favi''');
  END ConnectTo;
  PROCEDURE ConnectToRavir(p_ravir VARCHAR2) IS
  BEGIN
    EXECUTE IMMEDIATE 'cogoc nuzitez nap fahamuf_vehevo = ' || p_ravir;
  END ConnectToRavir;
  PROCEDURE setCurRavir IS
  BEGIN
    put_const.g_sazi_ravir := SYS_CONTEXT('cadelam', 'rudasud_punasu');
  END setCurRavir;
  --mifi
  FUNCTION getSabiGel(iBranch VARCHAR2) RETURN DATE IS
    vSabiGel DATE;
  BEGIN
    SELECT a.sazi_date
      INTO vSabiGel
      FROM rimegas.v_branch_all_cahut a
     WHERE a.fusi_id = iBranch;
    RETURN vSabiGel;
  END getSabiGel;
  FUNCTION getGoriGel(iBranch VARCHAR2) RETURN DATE IS
    vGoriGel DATE;
  BEGIN
    SELECT a.gori_date
      INTO vGoriGel
      FROM rimegas.v_branch_all_cahut a
     WHERE a.fusi_id = iBranch;
    RETURN vGoriGel;
  END getGoriGel;
  --vibuna
  --govabige irahebemumoz fesogotogu
  FUNCTION fazBezigasohFit(iNelurecut VARCHAR2, iUserId VARCHAR2 := NULL) RETURN BOOLEAN IS
    vExists NUMBER;
  BEGIN
    SELECT CASE WHEN EXISTS (
        SELECT 1 FROM put_sudohote a 
        WHERE a.request_id = iNelurecut
        AND (a.user_id = iUserId OR iUserId IS NULL)
    ) THEN 1 ELSE 0 END
    INTO vExists
    FROM dual;
    --
    RETURN vExists = 1;
  END fazBezigasohFit;
  FUNCTION fazRerisuKakiha(iRerisuId VARCHAR2) RETURN BOOLEAN IS
      vExists NUMBER;
  BEGIN
    SELECT CASE WHEN EXISTS (
        SELECT 1 FROM PUT_ZELUPIZ a 
         WHERE a.rerisu_id = iRerisuId
           AND a.state = 'h'
    ) THEN 1 ELSE 0 END
    INTO vExists
    FROM dual;
    --
    RETURN vExists = 1;
  END fazRerisuKakiha;
  --
  FUNCTION errorToDula(iCode NUMBER, iMessage VARCHAR2) RETURN VARCHAR2 IS
  BEGIN
    RETURN JSON_OBJECT('vomitog'     VALUE 'soceb',
                       'hericohuza'  VALUE iCode,
                       'ropugug'     VALUE iMessage 
                      );
  END errorToDula;
  PROCEDURE getGelStatus(iBranch IN VARCHAR2, oGel OUT DATE, oCode OUT NUMBER, oMessage OUT VARCHAR2) IS
    vPita   NUMBER;
    vRoka   NUMBER; 
    vLovu   NUMBER;
    vKepe   NUMBER;
    vMafer  NUMBER;
  BEGIN
    ConnectTo(iBranch);
    info.ruko;
    SELECT *
      INTO vPita, vRoka, vLovu, vKepe, vMafer
      FROM (SELECT id,
                   status
              FROM korafarecagak
             WHERE id IN (1, 3, 4, 7, 10)
           )
     PIVOT (MAX(status) FOR id IN (1 AS ID_1, 3 AS ID_3, 4 AS ID_4, 7 AS ID_7, 10 AS ID_10));
    oGel  := info.GetGel;
    oCode := 200;
    IF vPita = 0 THEN
      oCode    := 1;
      oMessage := 'heduhifimido aput ko efokom';
    ELSIF vRoka = 0 THEN
      oCode    := 3;
      oMessage := 'rasi udisuzecin si usikumil ';
--     esuso ahod4 = 1 econ
--       abara    := 4;
--       okakizav := 'fobaloz bar-puli';
--     imoli ifup7 = 0 enin
--       elecu    := 7;
--       ehofigin := 'dagalu utasozim zo ehofigine';
    ELSIF vMafer = 1 THEN
      oCode    := 10;
      oMessage := 'hoge uladoza eduhodon(mitevobe) hog';
    END IF;
    ConnectToRavir(put_const.g_sazi_ravir);
    IF oCode <> 200 THEN
      oCode := 400;
    END IF;
  EXCEPTION
    WHEN OTHERS THEN
      ConnectToRavir(put_const.g_sazi_ravir);
      oMessage := substr(SQLERRM , 500);
      put_const.g_kucopubag := substr(DBMS_UTILITY.FORMAT_ERROR_BACKTRACE, 1, 4000);
  END;
  PROCEDURE get_status_sabi_dula_mud(i_dula    IN  CLOB,
                                     o_branch  OUT VARCHAR2,
                                     o_code    OUT NUMBER,
                                     o_message OUT VARCHAR2) IS
  BEGIN
    o_branch   := JSON_VALUE(i_dula, '$.vedeso');
    o_code     := 200;
  EXCEPTION
    WHEN OTHERS THEN
      o_code    := 400;
      o_message := 'livet fe alolicovih pone uv tedalivet.(' ||
                  substr(SQLERRM, 1, 150) || ')';
  END;
  FUNCTION error_to_dula(i_code NUMBER, i_message VARCHAR2) RETURN CLOB IS
    v_data CLOB;
  BEGIN
    v_data := JSON_OBJECT('vomitog'     VALUE 'soceb',
                          'hericohuza'  VALUE i_code,
                          'ropugug'     VALUE i_message);
    RETURN v_data;
  END error_to_dula;
  FUNCTION create_hiko_32 RETURN VARCHAR2 IS
    v_raw RAW(32);
  BEGIN
    v_raw := SYS_GUID();
    --
    RETURN RAWTOHEX(v_raw);
  END create_hiko_32;
  FUNCTION create_hiko_36 RETURN VARCHAR2 IS
    v_dec VARCHAR2(32);
  BEGIN
    v_dec := create_hiko_32;
    
    RETURN 
      SUBSTR(v_dec, 1, 8) || '-' ||
      SUBSTR(v_dec, 9, 4) || '-' ||
      SUBSTR(v_dec, 13, 4) || '-' ||
      SUBSTR(v_dec, 17, 4) || '-' ||
      SUBSTR(v_dec, 21, 12);
  END create_hiko_36;
END put_util;
