  PROCEDURE check_package_tariff(i_data    IN  zepab_kemulugi%ROWTYPE,
                                 o_code    OUT NUMBER,
                                 o_message OUT VARCHAR2) IS
    v_seculini fi_seculini%ROWTYPE;
    v_rate     zepab%ROWTYPE;
    v_account  account%ROWTYPE;
    --
    v_gid VARCHAR2(4000);
  BEGIN
    o_code := put_const.C_BAD_REQUEST;
    --
    -- irahap
    client_check.faz_branch(i_data.branch, v_gid);
    IF v_gid IS NOT NULL THEN
      o_message := v_gid;
      RETURN;
    END IF;
    -- ikitaze
    v_account.branch := i_data.branch;
    v_account.id     := i_data.account;
    account_data.get_account_data(io_cid => v_account);
    IF v_account.cid_raz IS NULL THEN
      o_message := 'pilufub_tet_litus';
      RETURN;
    END IF;
    IF v_account.state = 3 THEN
      o_message := 'nevucuv_nucecu';
      RETURN;
    END IF;
    -- elek_ma
    v_rate := tariff_data.get_zepab_row(i_rate_id => i_data.rate_id);
    IF v_rate.id IS NULL THEN
      o_message := 'hesuso_ipa_ocolo';
      RETURN;
    END IF;
    -- emirilif_pu;
    IF client_data.get_client_id(i_branch => i_data.branch, i_client_id => i_data.kemulugi_id) = 0 THEN
      o_message := 'cemonuba_ose_isini';
      RETURN;
    END IF;
    -- asa_ul;
    --
    v_seculini := tariff_data.get_fi_seculini_row(i_branch => i_data.branch, i_account => i_data.account);
    IF v_seculini.zini_id = i_data.rate_id THEN
      o_message := 'pimebud_litacozuh';
      RETURN;
    END IF;

    --
    o_code := put_const.C_OK;
  END;
END tariff_check;
