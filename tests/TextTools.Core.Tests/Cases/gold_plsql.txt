PROCEDURE close_account(i_branch  IN VARCHAR2,
                        i_acc_id  IN account.id%TYPE,
                        o_code    OUT NUMBER,
                        o_message OUT VARCHAR2) IS
  v_acc account%ROWTYPE;
BEGIN
  v_acc.branch := i_branch;
  v_acc.id     := i_acc_id;
  account_data.get_account_data(io_acc => v_acc);
  IF v_acc.state <> account_const.c_state_active THEN
    o_code    := cbs_const.c_unprocessable;
    o_message := 'Account is not active';
    RETURN;
  END IF;
  account_kernel.close_account(i_data => v_acc);
  o_code := cbs_const.c_ok;
EXCEPTION
  WHEN NO_DATA_FOUND THEN
    o_code    := cbs_const.c_not_found;
    o_message := 'Account not found';
END close_account;
