BEGIN
  OPEN c FOR
    SELECT a.id
      FROM account a
     WHERE EXISTS (
       SELECT 1
         FROM client c
        WHERE c.id = a.client
          AND c.branch = i_branch
          AND c.state = client_const.c_state_active
          AND c.type_of_the_client = i_type_of_the_client -- the type
     );
END;
